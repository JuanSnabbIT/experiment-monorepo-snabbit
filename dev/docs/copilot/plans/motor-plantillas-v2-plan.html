import { Element, Transforms } from 'slate'; import type { ReactEditor } from 'slate-react'; import type { TNodoEtiqueta } from '@/interface/plantillaContratoV2.interface'; /** Plugin Slate: marca etiquetas y bloques transversales como nodos void */ export function withEtiquetas(editor: ReactEditor): ReactEditor { const { isVoid, isInline } = editor; editor.isVoid = (element: Element) => element.type === 'etiqueta' || element.type === 'bloque_transversal' ? true : isVoid(element); editor.isInline = (element: Element) => element.type === 'etiqueta' ? true : isInline(element); return editor; } /** Inserta un chip de etiqueta en la posición actual del cursor */ export function insertarEtiqueta(editor: ReactEditor, clave: string): void { const nodo: TNodoEtiqueta = { type: 'etiqueta', clave, children: [{ text: '' }] }; Transforms.insertNodes(editor, nodo); Transforms.move(editor); // mueve cursor después del chip }
