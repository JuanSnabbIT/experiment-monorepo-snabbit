    <AnimatePresence mode="wait">

            <motion.div
                key="search-field"
                className="flex gap-4"
                initial={{ opacity: 0, x: -50 }}
                animate={{ opacity: 1, x: 0 }}
                exit={{ opacity: 0, x: -50 }}
                transition={{ duration: 0.3 }}
            >
                <div>
                    
                </div>
                <FieldWrap
                    lastSuffix={
                        <Button
                        icon="HeroXMark"
                        color="red"
                        onClick={() => setBuscarAbierto(false)}
                        />
                    }
                >
                    <Input
                        className="max-w-[150px]"
                        name="globalFilter"
                        placeholder="Buscar..."
                        value={globalFilter}
                        onChange={(e) => setGlobalFilter(e.target.value)}
                    />
                </FieldWrap>
            </motion.div>
        ) : (
            <motion.div
                key="search-button"
                className="flex gap-4"
                initial={false}

                    variant="solid"
